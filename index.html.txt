  if (!isFinite(t) || t < 0) t = 0;
  const m = Math.floor(t/60);
  const s = Math.floor(t%60);
  const cs = Math.floor((t%1)*100);
  return m + ':' + String(s).padStart(2,'0') + '.' + String(cs).padStart(2,'0');
}

function probeVideo(url){
  return new Promise((resolve, reject) => {
    const v = document.createElement('video');
    v.preload = 'metadata';
    v.muted = true;
    v.playsInline = true;
    v.src = url;
    let done = false;
    const finish = (fn, arg) => { if (done) return; done = true; try{v.src='';}catch(e){} fn(arg); };
    v.onloadedmetadata = () => {
      const d = isFinite(v.duration) ? v.duration : 0;
      finish(resolve, { duration: d, width: v.videoWidth, height: v.videoHeight });
    };
    v.onerror = () => finish(reject, new Error('Unsupported or corrupted video'));
    setTimeout(() => finish(reject, new Error('Video metadata timeout')), 20000);
  });
}
function probeAudio(url){
  return new Promise((resolve, reject) => {
    const a = document.createElement('audio');
    a.preload = 'metadata';
    a.src = url;
    let done = false;
    const finish = (fn, arg) => { if (done) return; done = true; fn(arg); };
    a.onloadedmetadata = () => finish(resolve, { duration: isFinite(a.duration) ? a.duration : 0 });
    a.onerror = () => finish(reject, new Error('Unsupported or corrupted audio'));
    setTimeout(() => finish(reject, new Error('Audio metadata timeout')), 20000);
  });
}
function seekVideo(v, t){
  return new Promise((resolve) => {
    let done = false;
    const onSeek = () => { if (done) return; done = true; v.removeEventListener('seeked', onSeek); resolve(); };
    v.addEventListener('seeked', onSeek);
    try { v.currentTime = t; } catch(e){ resolve(); }
    setTimeout(() => { if (!done){ done = true; v.removeEventListener('seeked', onSeek); resolve(); } }, 2500);
  });
}
async function makeThumbs(url, duration, count = 6){
  const out = [];
  try {
    const v = document.createElement('video');
    v.muted = true; v.playsInline = true; v.preload = 'auto'; v.src = url;
    await new Promise((res, rej) => {
      v.onloadeddata = res;
      v.onerror = () => rej(new Error('thumb load fail'));
      setTimeout(rej, 15000);
    }).catch(()=>{ throw new Error('t'); });
    const c = document.createElement('canvas');
    c.width = 160; c.height = 90;
    const ctx = c.getContext('2d');
    const dur = duration > 0 ? duration : 1;
    for (let i = 0; i < count; i++){
      const t = dur * (i + 0.5) / count;
      await seekVideo(v, Math.min(t, Math.max(0, dur - 0.06)));
      ctx.drawImage(v, 0, 0, c.width, c.height);
      out.push(c.toDataURL('image/jpeg', 0.55));
    }
    try { v.src = ''; } catch(e){}
  } catch(e){ /* thumbnails optional */ }
  return out;
}

async function importFiles(files){
  if (!files || !files.length) return;
  let added = 0;
  let firstVideoAdded = false;

  for (const file of files){
    let type = null;
    if (file.type.startsWith('video')) type = 'video';
    else if (file.type.startsWith('audio')) type = 'audio';
    else if (file.type.startsWith('image')) type = 'image';
    else {
      const ext = (file.name.split('.').pop() || '').toLowerCase();
      if (['mp4','webm','mov','m4v','3gp','mkv','avi','ogv'].includes(ext)) type = 'video';
      else if (['mp3','wav','m4a','aac','ogg','oga','flac','opus'].includes(ext)) type = 'audio';
      else if (['jpg','jpeg','png','webp','gif','bmp','avif'].includes(ext)) type = 'image';
    }
    if (!type){ toast('Unsupported file: ' + file.name); continue; }

    const offline = state.media.find(m => m.offline && m.name === file.name && m.type === type);
    const url = URL.createObjectURL(file);

    if (offline){
      offline.url = url;
      offline.file = file;
      offline.offline = false;
      if (type === 'image'){ offline.imgEl = new Image(); offline.imgEl.src = url; }
      added++;
      if (!offline.thumb && type === 'image') offline.thumb = url;
      continue;
    }

    const item = {
      id: uid(), name: file.name, type, url, file,
      duration: 0, width: 0, height: 0, thumb: null, imgEl: null, offline: false
    };

    try {
      if (type === 'video'){
        const info = await probeVideo(url);
        item.duration = info.duration; item.width = info.width; item.height = info.height;
        item.thumbs = await makeThumbs(url, info.duration, 6);
        item.thumb = item.thumbs[0] || null;
        firstVideoAdded = true;
      } else if (type === 'audio'){
        const info = await probeAudio(url);
        item.duration = info.duration;
      } else {
        const img = new Image();
        await new Promise((res, rej) => {
          img.onload = res;
          img.onerror = () => rej(new Error('bad image'));
          setTimeout(rej, 12000);
          img.src = url;
        }).catch(()=>{ throw new Error('bad image'); });
        item.width = img.naturalWidth; item.height = img.naturalHeight;
        item.imgEl = img;
        item.thumb = url;
        item.duration = 4;
      }
      state.media.push(item);
      added++;
    } catch(err){
      URL.revokeObjectURL(url);
      toast('Could not load ' + file.name + ' — ' + (err.message || 'unsupported format'));
    }
  }
  renderMediaGrid();
  if (added) toast(added + ' file(s) imported');
  if (state.media.length && !state.project.video.length) $('#previewEmpty').style.display = 'none';
  checkCompat();

  if (firstVideoAdded || !state.baseFrame){
    const ok = await captureBaseFrame();
    if (ok){
      regenerateThumbnails();
    }
  }
}

function renderMediaGrid(){
  const g = $('#mediaGrid');
  g.innerHTML = '';
  if (!state.media.length){
    g.innerHTML = '<div class="hint" style="grid-column:1/-1">No media imported yet.</div>';
    return;
  }
  for (const m of state.media){
    const el = document.createElement('div');
    el.className = 'media-item';
    const icon = m.type === 'video' ? '🎞' : (m.type === 'audio' ? '🎵' : '🖼');
    el.innerHTML =
      '<div class="mi-thumb" style="' + (m.thumb ? 'background-image:url(' + m.thumb + ')' : '') + '">' + (m.thumb ? '' : icon) + '</div>' +
      '<div class="mi-badge">' + m.type + (m.offline ? ' · offline' : '') + '</div>' +
      '<div class="mi-name">' + escapeHtml(m.name) + '</div>' +
      '<div class="mi-meta">' + (m.offline ? 're-import required' : (m.type === 'image' ? (m.width + '×' + m.height) : fmtDur(m.duration))) + '</div>' +

      e.stopPropagation();

    });

  }
}
function removeMedia(m){
  for (const track of ['video','overlay','audio']){
    state.project[track] = state.project[track].filter(c => c.mediaId !== m.id);
  }
  if (m.url) URL.revokeObjectURL(m.url);
  state.media = state.media.filter(x => x.id !== m.id);
  pushHistory();
  refreshAll();
}

function addMediaToTimeline(m){
  if (m.offline){ toast('Media offline — re-import the file first.'); return; }
  if (m.type === 'video'){
    const start = state.project.video.reduce((a,c) => Math.max(a, c.start + clipDur(c)), 0);
    const c = {
      id: uid(), mediaId: m.id, start,
      inPoint: 0, outPoint: m.duration > 0 ? m.duration : 5,
      speed: 1, volume: 1, muted: false,
      transform: defaultTransform(), crop: defaultCrop(),
      color: defaultColor(), effects: defaultEffects(), curves: defaultCurves(),
      filter: 'original'
    };
    state.project.video.push(c);
    state.selected = { track:'video', id:c.id };
  } else if (m.type === 'audio'){
    const c = {
      id: uid(), mediaId: m.id, start: state.playhead,
      inPoint: 0, outPoint: m.duration > 0 ? m.duration : 5,
      volume: 1, muted: false, fadeIn: 0, fadeOut: 0, speed: 1
    };
    state.project.audio.push(c);
    state.selected = { track:'audio', id:c.id };
  } else {
    const c = {
      id: uid(), mediaId: m.id, start: state.playhead, duration: 4,
      opacity: 1, transform: defaultTransform(), crop: defaultCrop()
    };
    state.project.overlay.push(c);
    state.selected = { track:'overlay', id:c.id };
  }
  pushHistory();
  refreshAll();
  if (state.media.length) $('#previewEmpty').style.display = 'none';
}

/* ---------------- clip helpers ---------------- */
function clipDur(c){
  if (c.outPoint === undefined) return c.duration || 0;
  const sp = c.speed || 1;
  return Math.max(0, (c.outPoint - c.inPoint) / sp);
}
function clipEndOf(track, c){
  if (track === 'overlay' || track === 'text') return c.start + c.duration;
  return c.start + clipDur(c);
}
function projectDuration(){
  let d = 0;
  for (const c of state.project.video) d = Math.max(d, clipEndOf('video', c));
  for (const c of state.project.overlay) d = Math.max(d, clipEndOf('overlay', c));
  for (const c of state.project.text) d = Math.max(d, clipEndOf('text', c));
  for (const c of state.project.audio) d = Math.max(d, clipEndOf('audio', c));
  return d;
}
function clipAt(t){
  for (const c of state.project.video){
    if (t >= c.start - 1e-6 && t < c.start + clipDur(c) - 1e-6) return c;
  }
  return null;
}
function selClip(){
  if (!state.selected) return null;
  const arr = state.project[state.selected.track];
  if (!arr) return null;
  return arr.find(c => c.id === state.selected.id) || null;
}
function aspectValue(a){
  switch(a){
    case '9:16': return 9/16;
    case '4:3': return 4/3;
    case '1:1': return 1;
    case 'original': {
      const v = state.media.find(m => m.type === 'video' && m.width);
      if (v) return v.width / v.height;
      return 16/9;
    }
    default: return 16/9;
  }
}

    audioEls.delete(clipId);
  }
}
function audioVolumeAt(ac, local){
  let v = ac.volume;
  const dur = clipDur(ac);
  if (ac.fadeIn > 0 && local < ac.fadeIn) v *= clamp(local / ac.fadeIn, 0, 1);
  if (ac.fadeOut > 0 && local > dur - ac.fadeOut) v *= clamp((dur - local) / ac.fadeOut, 0, 1);
  return clamp(v, 0, 4);
}

/* ---------------- preview sizing ---------------- */
function resizePreview(){
  const stage = $('#previewStage');
  const rect = stage.getBoundingClientRect();
  if (rect.width < 10 || rect.height < 10) return;
  const aspect = aspectValue(state.project.aspect);
  let w = rect.width - 8, h = w / aspect;
  if (h > rect.height - 8){ h = rect.height - 8; w = h * aspect; }
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  let cw = Math.max(2, Math.round(w * dpr));
  let ch = Math.max(2, Math.round(h * dpr));
  const maxPx = 1280 * 720;
  if (cw * ch > maxPx){
    const k = Math.sqrt(maxPx / (cw * ch));
    cw = Math.max(2, Math.round(cw * k));
    ch = Math.max(2, Math.round(ch * k));
  }
  previewCanvas.width = cw;
  previewCanvas.height = ch;
  previewCanvas.style.width = Math.round(w) + 'px';
  previewCanvas.style.height = Math.round(h) + 'px';
  requestRender();
}

/* ---------------- render pipeline ---------------- */
let renderQueued = false;
function requestRender(){
  if (renderQueued) return;
  renderQueued = true;
  requestAnimationFrame(() => {
    renderQueued = false;
    if (!state.playing && !state.exporting) renderPreview();
  });
}

function ensureVideoFrame(t){
  const clip = clipAt(t);
  if (!clip) return null;
  const m = getMedia(clip.mediaId);
  if (!m || !m.url) return null;
  if (mainVideo.dataset.mediaId !== m.id){
    mainVideo.src = m.url;
    mainVideo.dataset.mediaId = m.id;
    try { mainVideo.load(); } catch(e){}
  }
  const target = clamp(clip.inPoint + (t - clip.start) * (clip.speed || 1), 0, Math.max(0, (m.duration || 0) - 0.03));
  mainVideo.playbackRate = clip.speed || 1;
  if (mainVideo.readyState >= 1 && Math.abs(mainVideo.currentTime - target) > 0.05 && !mainVideo.seeking){
    try { mainVideo.currentTime = target; } catch(e){}
  }
  return clip;
}

      $('#previewEmpty').style.display = '';

  renderFrame(ctx, previewCanvas.width, previewCanvas.height, state.playhead, true);
}

function cssFilterFromColor(c){
  const f = [];
  if (c.brightness) f.push('brightness(' + (1 + c.brightness/100).toFixed(3) + ')');
  if (c.contrast)   f.push('contrast(' + (1 + c.contrast/100).toFixed(3) + ')');
  if (c.saturation) f.push('saturate(' + (1 + c.saturation/100).toFixed(3) + ')');
  if (c.hue)        f.push('hue-rotate(' + c.hue + 'deg)');
  if (c.sepia)      f.push('sepia(' + (c.sepia/100).toFixed(3) + ')');
  if (c.grayscale)  f.push('grayscale(' + (c.grayscale/100).toFixed(3) + ')');
  if (c.blur)       f.push('blur(' + c.blur + 'px)');
  return f.length ? f.join(' ') : 'none';
}
function combinedCssFilter(clip){
  const parts = [];
  const f = FILTERS[clip.filter];
  if (f && f.css && f.css !== 'none') parts.push(f.css);
  const c = clip.color ? cssFilterFromColor(clip.color) : 'none';
  if (c && c !== 'none') parts.push(c);
  return parts.length ? parts.join(' ') : 'none';
}

/* ---------------- LUT ---------------- */
function buildCurveTable(pts){
  const p = pts.slice().sort((a,b) => a.x - b.x);
  const t = new Uint8ClampedArray(256);
  if (p.length < 2){ for (let i=0;i<256;i++) t[i] = i; return t; }
  for (let i = 0; i < 256; i++){
    let v;
    if (i <= p[0].x) v = p[0].y;
    else if (i >= p[p.length-1].x) v = p[p.length-1].y;
    else {
      let k = 0;
      while (k < p.length - 2 && i > p[k+1].x) k++;
      const a = p[k], b = p[k+1];
      const dx = (b.x - a.x) || 1;
      const u = clamp((i - a.x) / dx, 0, 1);
      const s = u*u*(3 - 2*u);
      v = a.y + (b.y - a.y) * s;
    }
    t[i] = clamp(Math.round(v), 0, 255);
  }
  return t;
}
function isIdentityCurve(curves){
  const id = (arr) => arr.length === 2 && arr[0].x === 0 && arr[0].y === 0 && arr[1].x === 255 && arr[1].y === 255;
  return id(curves.rgb) && id(curves.r) && id(curves.g) && id(curves.b);
}
function getLUT(clip){
  const c = clip.color, curves = clip.curves;
  const needTone = c.exposure || c.temperature || c.tint || c.highlights || c.shadows ||
                   c.whites || c.blacks || c.gamma;
  const needCurve = !isIdentityCurve(curves);
  if (!needTone && !needCurve) return null;

  const lut = new Uint8ClampedArray(768);
  const expo = Math.pow(2, c.exposure/100);
  const gamma = Math.pow(2, -c.gamma/100);
  const tempR = 1 + (c.temperature/100)*0.25;
  const tempB = 1 - (c.temperature/100)*0.25;
  const tempG = 1 + (c.tint/100)*0.16;
  const gains = [tempR, tempG, tempB];
  const sh = (c.shadows/100)*0.35;
  const hi = (c.highlights/100)*0.35;
  const bl = (c.blacks/100)*0.30;
  const wh = (c.whites/100)*0.30;

  const master = buildCurveTable(curves.rgb);
  const chans = [buildCurveTable(curves.r), buildCurveTable(curves.g), buildCurveTable(curves.b)];

  for (let ch = 0; ch < 3; ch++){
    for (let i = 0; i < 256; i++){
      let x = i / 255;
      x *= expo;
      const xc = clamp(x, 0, 1);
      x += sh * Math.pow(1 - xc, 3);
      x += hi * Math.pow(xc, 3);
      x += bl * Math.pow(1 - clamp(x,0,1), 5);
      x += wh * Math.pow(clamp(x,0,1), 5);
      x = clamp(x, 0, 1);
      if (gamma !== 1) x = Math.pow(x, gamma);
      x *= gains[ch];
      x = clamp(x, 0, 1);
      let v = x * 255;
      v = chans[ch][clamp(Math.round(v), 0, 255)];
      v = master[clamp(Math.round(v), 0, 255)];
      lut[ch*256 + i] = v;
    }
  }
  return lut;
}
function applyLUT(ctx, W, H, lut){
  let img;
  try { img = ctx.getImageData(0, 0, W, H); } catch(e){ return; }
  const d = img.data;
  for (let i = 0, n = d.length; i < n; i += 4){
    d[i]   = lut[d[i]];
    d[i+1] = lut[256 + d[i+1]];
    d[i+2] = lut[512 + d[i+2]];
  }
  ctx.putImageData(img, 0, 0);
}

/* ---------------- canvas helpers ---------------- */
function getScratch(W, H){
  if (scratch.width !== W || scratch.height !== H){
    scratch.width = W;
    scratch.height = H;
  }
  return scratch;
}
function applySharpen(ctx, W, H, k){
  if (k <= 0) return;
  let img;
  try { img = ctx.getImageData(0, 0, W, H); } catch(e){ return; }
  const s = new Uint8ClampedArray(img.data);
  const d = img.data;
  const c0 = 1 + 4*k, e = -k;
  const stride = W * 4;
  for (let y = 1; y < H - 1; y++){
    const r = y * W;
    for (let x = 1; x < W - 1; x++){
      const i = (r + x) * 4;
      for (let ch = 0; ch < 3; ch++){
        d[i+ch] = c0 * s[i+ch] + e * (s[i-4+ch] + s[i+4+ch] + s[i-stride+ch] + s[i+stride+ch]);
      }
    }
  }
  ctx.putImageData(img, 0, 0);
}
let noiseCanvas = null;
function getNoise(){
  if (noiseCanvas) return noiseCanvas;
  const c = document.createElement('canvas');
  c.width = c.height = 128;
  const ctx = c.getContext('2d');
  const img = ctx.createImageData(128, 128);
  const d = img.data;
  for (let i = 0; i < d.length; i += 4){
    const v = 128 + (Math.random() * 2 - 1) * 100;
    d[i] = d[i+1] = d[i+2] = v;
    d[i+3] = 255;
  }
  ctx.putImageData(img, 0, 0);
  noiseCanvas = c;
  return c;
}

/* ---------------- effects processing ---------------- */
function applyEffectToCtx(ctx, W, H, key, intensity){
  const k = clamp(intensity / 100, 0, 1);
  if (k <= 0) return;
  const minDim = Math.min(W, H);
  const REF = 360;
  const scale = minDim / REF;

  switch(key){
    case 'blur': {
      const radius = Math.max(0.5, k * 10 * scale);
      const sc = getScratch(W, H);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.filter = 'none';
      sctx.globalCompositeOperation = 'source-over';
      sctx.clearRect(0, 0, W, H);
      sctx.filter = 'blur(' + radius.toFixed(2) + 'px)';
      sctx.drawImage(ctx.canvas, 0, 0);
      sctx.filter = 'none';
      ctx.drawImage(sc, 0, 0);
      break;
    }
    case 'vignette': {
      const a = k * 0.85;
      const g = ctx.createRadialGradient(W/2, H/2, minDim * 0.30, W/2, H/2, Math.max(W,H) * 0.72);
      g.addColorStop(0, 'rgba(0,0,0,0)');
      g.addColorStop(1, 'rgba(0,0,0,' + a.toFixed(3) + ')');
      ctx.save();
      ctx.fillStyle = g;
      ctx.fillRect(0, 0, W, H);
      ctx.restore();
      break;
    }
    case 'grain': {
      ctx.save();
      ctx.globalAlpha = k * 0.30;
      ctx.globalCompositeOperation = 'overlay';
      const pat = ctx.createPattern(getNoise(), 'repeat');
      ctx.fillStyle = pat;
      ctx.fillRect(0, 0, W, H);
      ctx.restore();
      break;
    }
    case 'glow': {
      const sc = getScratch(W, H);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.globalCompositeOperation = 'source-over';
      sctx.filter = 'brightness(1.7) contrast(2.4)';
      sctx.clearRect(0, 0, W, H);
      sctx.drawImage(ctx.canvas, 0, 0);
      sctx.filter = 'none';
      const blurR = Math.max(2, Math.round(minDim * 0.02));
      ctx.save();
      ctx.globalCompositeOperation = 'lighter';
      ctx.globalAlpha = k * 0.65;
      ctx.filter = 'blur(' + blurR + 'px)';
      ctx.drawImage(sc, 0, 0);
      ctx.restore();
      ctx.filter = 'none';
      break;
    }
    case 'pixelate': {
      const px = Math.max(2, Math.round(k * minDim / 18) + 2);
      const sw = Math.max(1, Math.floor(W / px));
      const sh = Math.max(1, Math.floor(H / px));
      const sc = getScratch(sw, sh);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.filter = 'none';
      sctx.imageSmoothingEnabled = false;
      sctx.clearRect(0, 0, sw, sh);
      sctx.drawImage(ctx.canvas, 0, 0, W, H, 0, 0, sw, sh);
      ctx.imageSmoothingEnabled = false;
      ctx.drawImage(sc, 0, 0, sw, sh, 0, 0, W, H);
      ctx.imageSmoothingEnabled = true;
      break;
    }
    case 'rgbShift': {
      const off = Math.max(1, Math.round(k * 15 * scale));
      let img;
      try { img = ctx.getImageData(0, 0, W, H); } catch(e){ return; }
      const d = img.data;
      const src = new Uint8ClampedArray(d);
      for (let y = 0; y < H; y++){
        const row = y * W;
        for (let x = 0; x < W; x++){
          const i = (row + x) * 4;
          const xr = clamp(x + off, 0, W - 1);
          const xb = clamp(x - off, 0, W - 1);
          d[i]   = src[(row + xr) * 4];
          d[i+2] = src[(row + xb) * 4 + 2];
        }
      }
      ctx.putImageData(img, 0, 0);
      break;
    }
    case 'mirror': {
      const halfW = Math.floor(W / 2);
      if (halfW < 2) return;
      const sc = getScratch(halfW, H);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.filter = 'none';
      sctx.clearRect(0, 0, halfW, H);
      sctx.drawImage(ctx.canvas, 0, 0, halfW, H, 0, 0, halfW, H);
      ctx.save();
      ctx.globalAlpha = k;
      ctx.translate(W, 0);
      ctx.scale(-1, 1);
      ctx.drawImage(sc, 0, 0, halfW, H, 0, 0, halfW, H);
      ctx.restore();
      break;
    }
    case 'fade': {
      ctx.save();
      ctx.globalAlpha = k * 0.5;
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, W, H);
      ctx.restore();
      break;
    }
    case 'bw': {
      const sc = getScratch(W, H);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.globalCompositeOperation = 'source-over';
      sctx.filter = 'grayscale(1)';
      sctx.clearRect(0, 0, W, H);
      sctx.drawImage(ctx.canvas, 0, 0);
      sctx.filter = 'none';
      ctx.save();
      ctx.globalAlpha = k;
      ctx.drawImage(sc, 0, 0);
      ctx.restore();
      break;
    }
    case 'sepia': {
      const sc = getScratch(W, H);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.globalCompositeOperation = 'source-over';
      sctx.filter = 'sepia(1)';
      sctx.clearRect(0, 0, W, H);
      sctx.drawImage(ctx.canvas, 0, 0);
      sctx.filter = 'none';
      ctx.save();
      ctx.globalAlpha = k;
      ctx.drawImage(sc, 0, 0);
      ctx.restore();
      break;
    }
    case 'sharpen': {
      applySharpen(ctx, W, H, k * 0.7);
      break;
    }
    case 'zoom': {
      const z = 1 + k * 0.5;
      const sc = getScratch(W, H);
      const sctx = sc.getContext('2d');
      sctx.setTransform(1,0,0,1,0,0);
      sctx.filter = 'none';
      sctx.globalCompositeOperation = 'source-over';
      sctx.clearRect(0, 0, W, H);
      sctx.drawImage(ctx.canvas, 0, 0);
      ctx.save();
      ctx.setTransform(1,0,0,1,0,0);
      ctx.globalCompositeOperation = 'source-over';
      ctx.globalAlpha = 1;
      ctx.filter = 'none';
      ctx.clearRect(0, 0, W, H);
      ctx.translate(W/2, H/2);
      ctx.scale(z, z);
      ctx.drawImage(sc, -W/2, -H/2);
      ctx.restore();
      break;
    }
  }
}
function applyEffectsToCtx(ctx, W, H, effects){
  if (!effects) return;
  for (const key of EFFECT_ORDER){
    const v = effects[key] || 0;
    if (v > 0){
      try { applyEffectToCtx(ctx, W, H, key, v); } catch(e){}
    }
  }
}

/* ---------------- video + overlay + text drawing ---------------- */
function drawVideoClip(ctx, W, H, clip){
  const tr = clip.transform, crop = clip.crop;
  const sw = mainVideo.videoWidth, sh = mainVideo.videoHeight;
  if (!sw || !sh) return;

  const cx = sw * crop.left / 100;
  const cy = sh * crop.top / 100;
  const cw = Math.max(1, sw * (1 - (crop.left + crop.right) / 100));
  const ch = Math.max(1, sh * (1 - (crop.top + crop.bottom) / 100));

  const fitCover = tr.fit === 'cover';
  const base = fitCover ? Math.max(W / cw, H / ch) : Math.min(W / cw, H / ch);
  const scale = base * (tr.scale || 1);

  ctx.save();
  ctx.filter = combinedCssFilter(clip);
  ctx.globalAlpha = 1;
  ctx.globalCompositeOperation = 'source-over';
  ctx.translate(W/2 + (tr.x || 0) * W / 2, H/2 + (tr.y || 0) * H / 2);
  if (tr.rotate) ctx.rotate(tr.rotate * Math.PI / 180);
  ctx.scale(tr.flipH ? -1 : 1, tr.flipV ? -1 : 1);
  const dw = cw * scale, dh = ch * scale;
  try {
    ctx.drawImage(mainVideo, cx, cy, cw, ch, -dw/2, -dh/2, dw, dh);
  } catch(e){}
  ctx.restore();
  ctx.filter = 'none';
}

function drawOverlayClip(ctx, W, H, oc, img){
  const tr = oc.transform;
  const sw = img.naturalWidth || img.width, sh = img.naturalHeight || img.height;
  if (!sw || !sh) return;
  const crop = oc.crop || {top:0,right:0,bottom:0,left:0};
  const cx = sw * crop.left / 100, cy = sh * crop.top / 100;
  const cw = Math.max(1, sw * (1 - (crop.left + crop.right) / 100));
  const ch = Math.max(1, sh * (1 - (crop.top + crop.bottom) / 100));
  const fitCover = tr.fit === 'cover';
  const base = fitCover ? Math.max(W / cw, H / ch) : Math.min(W / cw, H / ch);
  const scale = base * (tr.scale || 1) * 0.7;
  ctx.save();
  ctx.globalAlpha = oc.opacity !== undefined ? oc.opacity : 1;
  ctx.translate(W/2 + (tr.x || 0) * W / 2, H/2 + (tr.y || 0) * H / 2);
  if (tr.rotate) ctx.rotate(tr.rotate * Math.PI / 180);
  ctx.scale(tr.flipH ? -1 : 1, tr.flipV ? -1 : 1);
  try { ctx.drawImage(img, cx, cy, cw, ch, -cw*scale/2, -ch*scale/2, cw*scale, ch*scale); } catch(e){}
  ctx.restore();
}

function textFont(tc, H){
  const px = Math.max(4, tc.size * (H / 720));
  return { px, font: (tc.weight || '700') + ' ' + px + 'px ' + (tc.font || 'system-ui,-apple-system,Segoe UI,Roboto,sans-serif') };
}
function textBBox(tc, W, H){
  const f = textFont(tc, H);
  measureCtx.font = f.font;
  const lines = String(tc.text || '').split('\n');
  let maxW = 0;
  for (const l of lines) maxW = Math.max(maxW, measureCtx.measureText(l).width);
  const lh = f.px * 1.2;
  const h = lh * lines.length;
  const cx = tc.x * W, cy = tc.y * H;
  let x0 = cx;
  if (tc.align === 'center') x0 = cx - maxW / 2;
  else if (tc.align === 'right') x0 = cx - maxW;
  return { x: x0, y: cy - h/2, w: Math.max(4, maxW), h: Math.max(4, h), px: f.px };
}
function drawTextClip(ctx, W, H, tc){
  const f = textFont(tc, H);
  const lines = String(tc.text || '').split('\n');
  const lh = f.px * 1.2;
  const totalH = lh * lines.length;
  ctx.save();
  ctx.globalAlpha = clamp(tc.opacity, 0, 1);
  ctx.translate(tc.x * W, tc.y * H);
  ctx.rotate((tc.rotation || 0) * Math.PI / 180);
  ctx.font = f.font;
  ctx.textAlign = tc.align || 'center';
  ctx.textBaseline = 'middle';

  if (tc.bg && tc.bg !== 'transparent'){
    measureCtx.font = f.font;
    let maxW = 0;
    for (const l of lines) maxW = Math.max(maxW, measureCtx.measureText(l).width);
    const padX = f.px * 0.28, padY = f.px * 0.18;
    let bx = 0;
    if (tc.align === 'center') bx = -maxW / 2;
    else if (tc.align === 'right') bx = -maxW;
    ctx.fillStyle = tc.bg;
    ctx.fillRect(bx - padX, -totalH/2 - padY, maxW + padX*2, totalH + padY*2);
  }
  ctx.fillStyle = tc.color || '#fff';
  for (let i = 0; i < lines.length; i++){
    ctx.fillText(lines[i], 0, -totalH/2 + lh * (i + 0.5));
  }
  ctx.restore();
}

function renderFrame(ctx, W, H, t, isPreview){
  ctx.setTransform(1,0,0,1,0,0);
  ctx.globalAlpha = 1;
  ctx.globalCompositeOperation = 'source-over';
  ctx.filter = 'none';
  ctx.fillStyle = '#000';
  ctx.fillRect(0, 0, W, H);

  const vclip = clipAt(t);

  if (vclip && mainVideo.readyState >= 2 && mainVideo.videoWidth){
    drawVideoClip(ctx, W, H, vclip);
    const lut = getLUT(vclip);
    if (lut) applyLUT(ctx, W, H, lut);
    if (vclip.color && vclip.color.sharpness > 0){
      applySharpen(ctx, W, H, vclip.color.sharpness / 100 * 0.7);
    }
    applyEffectsToCtx(ctx, W, H, vclip.effects);
  }

  for (const oc of state.project.overlay){
    if (t < oc.start || t >= oc.start + oc.duration) continue;
    const m = getMedia(oc.mediaId);
    if (!m || !m.imgEl || !m.imgEl.complete || !m.imgEl.naturalWidth) continue;
    drawOverlayClip(ctx, W, H, oc, m.imgEl);
  }
  for (const tc of state.project.text){
    if (t < tc.start || t >= tc.start + tc.duration) continue;
    drawTextClip(ctx, W, H, tc);
  }

  if (isPreview && state.selected && state.selected.track === 'video'){
    const sc = selClip();
    if (sc && sc.crop && (sc.crop.top || sc.crop.bottom || sc.crop.left || sc.crop.right)){
      ctx.save();
      ctx.strokeStyle = 'rgba(91,140,255,.85)';
      ctx.setLineDash([6, 5]);
      ctx.lineWidth = 1.5;
      const l = W * sc.crop.left / 100, r = W * (1 - sc.crop.right / 100);
      const tp = H * sc.crop.top / 100, b = H * (1 - sc.crop.bottom / 100);
      ctx.strokeRect(l, tp, r - l, b - tp);
      ctx.restore();
    }
  }

  if (isPreview) drawSelectionOverlay(ctx, W, H, t);
}

function drawSelectionOverlay(ctx, W, H, t){
  if (!state.selected) return;
  const track = state.selected.track;
  const c = selClip();
  if (!c) return;
  if (track === 'text'){
    const bb = textBBox(c, W, H);
    ctx.save();
    ctx.strokeStyle = '#5b8cff';
    ctx.lineWidth = 1.5;
    ctx.setLineDash([5,4]);
    ctx.strokeRect(bb.x - 4, bb.y - 4, bb.w + 8, bb.h + 8);
    ctx.restore();
  } else if (track === 'overlay' && t >= c.start && t < c.start + c.duration){
    const tr = c.transform;
    const m = getMedia(c.mediaId);
    if (!m) return;
    const sw = m.width || 1, sh = m.height || 1;
    const fitCover = tr.fit === 'cover';
    const base = fitCover ? Math.max(W/sw, H/sh) : Math.min(W/sw, H/sh);
    const scale = base * (tr.scale || 1) * 0.7;
    const w = sw * scale, h = sh * scale;
    const cx = W/2 + (tr.x || 0) * W/2, cy = H/2 + (tr.y || 0) * H/2;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate((tr.rotate||0) * Math.PI / 180);
    ctx.strokeStyle = '#5b8cff';
    ctx.lineWidth = 1.5;
    ctx.setLineDash([5,4]);
    ctx.strokeRect(-w/2, -h/2, w, h);
    ctx.setLineDash([]);
    ctx.fillStyle = '#5b8cff';
    ctx.beginPath();
    ctx.arc(w/2, h/2, 9, 0, Math.PI*2);
    ctx.fill();
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(w/2, h/2, 3, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }
}

/* ---------------- base frame & thumbnails ---------------- */
async function captureBaseFrame(){
  let source = null;
  if (state.selected && state.selected.track === 'video'){
    const c = selClip();
    if (c){
      const m = getMedia(c.mediaId);
      if (m && m.url) source = m;
    }
  }
  if (!source){
    const c = state.project.video[0];
    if (c){
      const m = getMedia(c.mediaId);
      if (m && m.url) source = m;
    }
  }
  if (!source){
    source = state.media.find(m => m.type === 'video' && m.url && !m.offline);
  }
  if (!source){
    source = state.media.find(m => m.type === 'image' && m.url && m.imgEl);
  }
  if (!source) return false;

  try {
    if (source.type === 'image'){
      const img = source.imgEl;
      if (!img || !img.naturalWidth) return false;
      const sw = img.naturalWidth, sh = img.naturalHeight;
      const tw = 160;
      const th = Math.max(1, Math.round(tw * sh / sw));
      const c = document.createElement('canvas');
      c.width = tw; c.height = th;
      c.getContext('2d').drawImage(img, 0, 0, tw, th);
      state.baseFrame = { canvas: c, dataUrl: c.toDataURL('image/jpeg', 0.78) };
      return true;
    }
    const v = document.createElement('video');
    v.muted = true;
    v.playsInline = true;
    v.preload = 'auto';
    v.src = source.url;
    try {
      await new Promise((res, rej) => {
        v.onloadeddata = res;
        v.onerror = () => rej(new Error('load fail'));
        setTimeout(() => rej(new Error('timeout')), 15000);
      });
      const dur = isFinite(v.duration) ? v.duration : 0;
      const t = dur > 0.4 ? Math.max(0.1, Math.min(dur * 0.2, dur - 0.1)) : 0;
      if (t > 0) await seekVideo(v, t);
      const sw = v.videoWidth, sh = v.videoHeight;
      if (!sw || !sh) throw new Error('no video dimensions');
      const tw = 160;
      const th = Math.max(1, Math.round(tw * sh / sw));
      const c = document.createElement('canvas');
      c.width = tw; c.height = th;
      c.getContext('2d').drawImage(v, 0, 0, tw, th);
      state.baseFrame = { canvas: c, dataUrl: c.toDataURL('image/jpeg', 0.78) };
      return true;
    } finally {
      try { v.src = ''; } catch(e){}
    }
  } catch(e){
    return false;
  }
}
function generateEffectThumbs(){
  if (!state.baseFrame) return;
  const base = state.baseFrame.canvas;
  const W = base.width, H = base.height;
  const out = { none: state.baseFrame.dataUrl };
  for (const key of Object.keys(EFFECTS)){
    if (key === 'none') continue;
    const c = document.createElement('canvas');
    c.width = W; c.height = H;
    const ctx = c.getContext('2d');
    ctx.drawImage(base, 0, 0);
    try {
      applyEffectToCtx(ctx, W, H, key, EFFECTS[key].def || 60);
      out[key] = c.toDataURL('image/jpeg', 0.78);
    } catch(e){
      out[key] = state.baseFrame.dataUrl;
    }
  }
  state.effectThumbs = out;
}

}

/* ---------------- playback ---------------- */
function play(){
  const dur = projectDuration();
  if (dur <= 0){ toast('Add media to the timeline first'); return; }
  if (state.playhead >= dur - 0.03) state.playhead = 0;
  state.playing = true;
  state.lastTs = 0;
  state.activeClipId = null;
  $('#btnPlay').textContent = '❚❚';
  requestAnimationFrame(loop);
  syncAudioForce(true);
}
function pause(){
  if (!state.playing) return;
  state.playing = false;
  $('#btnPlay').textContent = '▶';
  try { mainVideo.pause(); } catch(e){}
  for (const el of audioEls.values()){ try { el.pause(); } catch(e){} }
}
function stop(){
  pause();
  state.playhead = 0;
  state.activeClipId = null;
  try { mainVideo.currentTime = 0; } catch(e){}
  updatePlayheadUI();
  requestRender();
}
function loop(ts){
  if (!state.playing) return;
  if (!state.lastTs){ state.lastTs = ts; requestAnimationFrame(loop); return; }
  const dt = Math.min(0.25, (ts - state.lastTs) / 1000);
  state.lastTs = ts;

  const dur = projectDuration();
  if (dur <= 0){ pause(); return; }

  state.playhead += dt;
  if (state.playhead >= dur){
    state.playhead = dur;
    pause();
    updatePlayheadUI();
    renderPreview();
    return;
  }

  const clip = clipAt(state.playhead);
  if (!clip){
    if (state.activeClipId){ state.activeClipId = null; try { mainVideo.pause(); } catch(e){} }
  } else {
    if (state.activeClipId !== clip.id){
      state.activeClipId = clip.id;
      loadClipForPlayback(clip, state.playhead);
    } else {
      const target = clip.inPoint + (state.playhead - clip.start) * (clip.speed || 1);
      if (!mainVideo.seeking && Math.abs(mainVideo.currentTime - target) > 0.32){
        try { mainVideo.currentTime = target; } catch(e){}
      }
      if (mainVideo.paused){ mainVideo.play().catch(()=>{}); }
    }
  }

  syncAudioForce(false);
  updatePlayheadUI();
  renderPreview();
  requestAnimationFrame(loop);
}
function loadClipForPlayback(clip, t){
  const m = getMedia(clip.mediaId);
  if (!m || !m.url) return;
  if (mainVideo.dataset.mediaId !== m.id){
    mainVideo.src = m.url;
    mainVideo.dataset.mediaId = m.id;
  }
  mainVideo.playbackRate = clip.speed || 1;
  mainVideo.volume = clamp((clip.muted ? 0 : clip.volume) * state.project.originalVolume, 0, 1);
  mainVideo.muted = state.project.originalMuted || clip.muted;
  const target = clamp(clip.inPoint + (t - clip.start) * (clip.speed || 1), 0, Math.max(0, (m.duration || 0) - 0.03));
  if (mainVideo.readyState >= 1){
    try { mainVideo.currentTime = target; } catch(e){}
  } else {
    mainVideo.addEventListener('loadedmetadata', () => {
      try { mainVideo.currentTime = target; } catch(e){}
    }, { once:true });
  }
  mainVideo.play().catch(()=>{});
}
let audioSyncTick = 0;
function syncAudioForce(force){
  audioSyncTick++;
  const now = state.playhead;
  const playing = state.playing;
  for (const ac of state.project.audio){
    const m = getMedia(ac.mediaId);
    if (!m || !m.url) continue;
    const el = getAudioEl(ac);
    const t0 = ac.start;
    const dur = clipDur(ac);
    const t1 = t0 + dur;
    if (playing && now >= t0 && now < t1){
      const local = now - t0;
      const target = clamp(ac.inPoint + local * (ac.speed || 1), 0, Math.max(0, (m.duration || 0) - 0.03));
      if (el.paused){
        try { el.currentTime = target; } catch(e){}
        el.play().catch(()=>{});
      } else if ((force || audioSyncTick % 20 === 0) && !el.seeking &&
                 Math.abs(el.currentTime - target) > 0.3){
        try { el.currentTime = target; } catch(e){}
      }
      el.playbackRate = ac.speed || 1;
      el.volume = clamp(audioVolumeAt(ac, local), 0, 1);
      el.muted = !!ac.muted;
    } else {
      if (!el.paused){ try { el.pause(); } catch(e){} }
    }
  }
  const vc = clipAt(now);
  if (vc){
    mainVideo.volume = clamp((vc.muted ? 0 : vc.volume) * state.project.originalVolume, 0, 1);
    mainVideo.muted = state.project.originalMuted || vc.muted;
  }
}

/* ---------------- timeline rendering ---------------- */
function renderTimeline(){
  const dur = Math.max(projectDuration(), 2);
  const zoom = state.zoom;
  const contentW = PAD + dur * zoom + 260;
  $('#tlContent').style.width = contentW + 'px';

  const ruler = $('#tlRuler');
  ruler.innerHTML = '';
  const steps = [0.1,0.2,0.5,1,2,5,10,15,30,60,120,300,600,1800];
  let step = steps.find(s => s * zoom >= 64) || 1800;
  for (let t = 0; t <= dur + step; t += step){
    const d = document.createElement('div');
    d.className = 'tick';
    d.style.left = (PAD + t * zoom) + 'px';
    d.innerHTML = '<span>' + fmtShort(t) + '</span>';
    ruler.appendChild(d);
  }

  for (const track of ['video','overlay','text','audio']){
    const lane = $('.tl-lane[data-track="' + track + '"]');
    lane.innerHTML = '';
    for (const c of state.project[track]){
      lane.appendChild(buildClipEl(track, c));
    }
  }
  updatePlayheadUI();
  $('#selLabel').textContent = state.selected
    ? (state.selected.track + ' clip selected')
    : 'No clip selected';
}

function buildClipEl(track, c){
  const zoom = state.zoom;
  const el = document.createElement('div');
  el.className = 'clip ' + track;
  if (state.selected && state.selected.id === c.id && state.selected.track === track) el.classList.add('sel');

  const dur = (track === 'overlay' || track === 'text') ? c.duration : clipDur(c);
  el.style.left = (PAD + c.start * zoom) + 'px';
  el.style.width = Math.max(14, dur * zoom) + 'px';

  const m = getMedia(c.mediaId);
  if (track === 'video' && m && m.thumbs && m.thumbs.length){
    el.style.backgroundImage = 'url(' + m.thumbs[0] + ')';
    el.style.backgroundSize = 'auto 100%';
    el.style.backgroundRepeat = 'repeat-x';
  }
  if (track === 'overlay' && m && m.thumb){
    el.style.backgroundImage = 'url(' + m.thumb + ')';
    el.style.backgroundSize = 'auto 100%';
    el.style.backgroundRepeat = 'repeat-x';
  }

  const cap = document.createElement('div');
  cap.className = 'cap';
  if (track === 'text') cap.textContent = (c.text || '').split('\n')[0].slice(0, 40) || 'Text';
  else cap.textContent = m ? m.name : 'Media';
  el.appendChild(cap);

  if (track === 'video' || track === 'audio'){
    const hl = document.createElement('div');
    hl.className = 'handle l';
    const hr = document.createElement('div');
    hr.className = 'handle r';
    el.appendChild(hl); el.appendChild(hr);
    hl.addEventListener('pointerdown', (e) => startClipDrag(e, el, track, c, 'left'));
    hr.addEventListener('pointerdown', (e) => startClipDrag(e, el, track, c, 'right'));
  }

  el.addEventListener('pointerdown', (e) => {
    if (e.target.classList.contains('handle')) return;
    selectClip(track, c.id);
    startClipDrag(e, el, track, c, 'move');
  });

  return el;
}

function layoutClipEl(el, track, c){
  const zoom = state.zoom;
  const dur = (track === 'overlay' || track === 'text') ? c.duration : clipDur(c);
  el.style.left = (PAD + c.start * zoom) + 'px';
  el.style.width = Math.max(14, dur * zoom) + 'px';
}

function startClipDrag(e, el, track, c, mode){
  if (e.button !== undefined && e.button !== 0 && e.pointerType === 'mouse') return;
  e.preventDefault();
  e.stopPropagation();
  const startX = e.clientX;
  const s0 = c.start, i0 = c.inPoint || 0, o0 = c.outPoint || 0;
  const m = getMedia(c.mediaId);
  const mediaDur = m && m.duration ? m.duration : 3600;
  el.setPointerCapture && el.setPointerCapture(e.pointerId);

  const move = (ev) => {
    const dx = (ev.clientX - startX) / state.zoom;
    if (mode === 'move'){
      c.start = Math.max(0, s0 + dx);
    } else if (mode === 'left'){
      const sp = c.speed || 1;
      const maxShift = clipDur(c) - 0.1;
      let shift = clamp(dx, -s0, maxShift);
      const newIn = clamp(i0 + shift * sp, 0, o0 - 0.1);
      shift = (newIn - i0) / sp;
      c.start = Math.max(0, s0 + shift);
      c.inPoint = newIn;
    } else if (mode === 'right'){
      const sp = c.speed || 1;
      c.outPoint = clamp(o0 + dx * sp, i0 + 0.1, mediaDur);
    } else if (mode === 'dur'){
      c.duration = Math.max(0.2, c.duration + dx);
    }
    layoutClipEl(el, track, c);
  };
  const up = () => {
    el.removeEventListener('pointermove', move);
    el.removeEventListener('pointerup', up);
    el.removeEventListener('pointercancel', up);
    pushHistory();
    refreshAll();
  };
  el.addEventListener('pointermove', move);
  el.addEventListener('pointerup', up);
  el.addEventListener('pointercancel', up);
}

function selectClip(track, id){
  state.selected = { track, id };
  refreshAll();
}

/* ---------------- timeline seeking ---------------- */
function initTimelineSeeking(){
  const content = $('#tlContent');
  const scroll = $('#tlScroll');

  const seekFromEvent = (clientX) => {
    const rect = content.getBoundingClientRect();
    const x = clientX - rect.left;
    const t = Math.max(0, (x - PAD) / state.zoom);
    state.playhead = Math.min(t, Math.max(projectDuration(), 0));
    updatePlayheadUI();
    if (!state.playing){
      state.activeClipId = null;
      requestRender();
    }
  };

  let seeking = false;
  $('#tlRuler').addEventListener('pointerdown', (e) => {
    seeking = true;
    $('#tlRuler').setPointerCapture && $('#tlRuler').setPointerCapture(e.pointerId);
    seekFromEvent(e.clientX);
  });
  $('#tlRuler').addEventListener('pointermove', (e) => { if (seeking) seekFromEvent(e.clientX); });
  $('#tlRuler').addEventListener('pointerup', () => { seeking = false; });
  $('#tlRuler').addEventListener('pointercancel', () => { seeking = false; });

  for (const lane of $$('.tl-lane')){
    lane.addEventListener('pointerdown', (e) => {
      if (e.target !== lane) return;
      seekFromEvent(e.clientX);
    });
  }

  scroll.addEventListener('wheel', (e) => {
    if (e.ctrlKey || e.metaKey){
      e.preventDefault();
      setZoom(state.zoom * (e.deltaY < 0 ? 1.15 : 0.87));
    }
  }, { passive:false });
}

}

function getClipParam(clip, key){
  if (clip.color && key in clip.color) return clip.color[key];
  if (clip.effects && key in clip.effects) return clip.effects[key];
  return 0;
}
function setClipParam(clip, key, value){
  if (clip.color && key in clip.color) clip.color[key] = value;
  else if (clip.effects && key in clip.effects) clip.effects[key] = value;
}

  }

function currentCurvesClip(){
  const c = selClip();
  if (!c || state.selected.track !== 'video') return null;
  return c;
}
function drawCurveGraph(){
  const cv = $('#curveCanvas');
  const ctx = cv.getContext('2d');
  const W = cv.width, H = cv.height;
  ctx.setTransform(1,0,0,1,0,0);
  ctx.fillStyle = '#0d0d11';
  ctx.fillRect(0, 0, W, H);

  ctx.strokeStyle = '#1e1e28';
  ctx.lineWidth = 1;
  for (let i = 1; i < 4; i++){
    const x = W * i / 4, y = H * i / 4;
    ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke();
  }
  ctx.strokeStyle = '#2c2c3a';
  ctx.beginPath(); ctx.moveTo(0, H); ctx.lineTo(W, 0); ctx.stroke();

  const c = currentCurvesClip();
  if (!c){
    ctx.fillStyle = '#55556a';
    ctx.font = '500 26px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('Select a video clip', W/2, H/2);
    return;
  }
  const pts = (c.curves[state.activeCurve] || []).slice().sort((a,b) => a.x - b.x);
  const tbl = buildCurveTable(pts);

  const colors = { rgb:'#ffffff', r:'#ff5d6c', g:'#4ddb8b', b:'#5b8cff' };
  ctx.strokeStyle = colors[state.activeCurve];
  ctx.lineWidth = 2.5;
  ctx.beginPath();
  for (let i = 0; i < 256; i++){
    const x = i / 255 * W;
    const y = H - tbl[i] / 255 * H;
    if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
  }
  ctx.stroke();

  for (const p of pts){
    const x = p.x / 255 * W, y = H - p.y / 255 * H;
    ctx.beginPath();
    ctx.arc(x, y, 9, 0, Math.PI * 2);
    ctx.fillStyle = '#131319';
    ctx.fill();
    ctx.strokeStyle = colors[state.activeCurve];
    ctx.lineWidth = 3;
    ctx.stroke();
  }
}
function curvePointerPos(e, cv){
  const r = cv.getBoundingClientRect();
  return {
    x: clamp((e.clientX - r.left) / r.width, 0, 1) * 255,
    y: clamp(1 - (e.clientY - r.top) / r.height, 0, 1) * 255
  };
}
let curveDrag = null;
function initCurveEditor(){
  const cv = $('#curveCanvas');
  cv.addEventListener('pointerdown', (e) => {
    const c = currentCurvesClip();
    if (!c){ toast('Select a video clip first'); return; }
    e.preventDefault();
    cv.setPointerCapture && cv.setPointerCapture(e.pointerId);
    const pos = curvePointerPos(e, cv);
    const pts = c.curves[state.activeCurve];
    let hit = -1, hitDist = 1e9;
    for (let i = 0; i < pts.length; i++){
      const d = Math.hypot(pts[i].x - pos.x, pts[i].y - pos.y);
      if (d < hitDist){ hitDist = d; hit = i; }
    }
    const tol = 255 * (18 / cv.getBoundingClientRect().width);
    if (hit >= 0 && hitDist <= tol){
      curveDrag = { index: hit, clip: c };
    } else {
      pts.push({ x: pos.x, y: pos.y });
      pts.sort((a,b) => a.x - b.x);
      const idx = pts.findIndex(p => p.x === pos.x && p.y === pos.y);
      curveDrag = { index: idx, clip: c };
    }
    drawCurveGraph();
    requestRender();
  });
  cv.addEventListener('pointermove', (e) => {
    if (!curveDrag) return;
    e.preventDefault();
    const c = curveDrag.clip;
    const pts = c.curves[state.activeCurve];
    const idx = curveDrag.index;
    if (idx < 0 || idx >= pts.length) return;
    const pos = curvePointerPos(e, cv);
    const isFirst = idx === 0;
    const isLast = idx === pts.length - 1;
    if (isFirst) pts[idx].x = 0;
    else if (isLast) pts[idx].x = 255;
    else {
      const lo = pts[idx-1].x + 2;
      const hi = pts[idx+1].x - 2;
      pts[idx].x = clamp(pos.x, lo, hi);
    }
    pts[idx].y = clamp(pos.y, 0, 255);
    drawCurveGraph();
    requestRender();
  });
  const endDrag = () => { if (curveDrag){ curveDrag = null; pushHistory(); } };
  cv.addEventListener('pointerup', endDrag);
  cv.addEventListener('pointercancel', endDrag);
  cv.addEventListener('dblclick', (e) => {
    const c = currentCurvesClip();
    if (!c) return;
    const pos = curvePointerPos(e, cv);
    const pts = c.curves[state.activeCurve];
    let hit = -1, hitDist = 1e9;
    for (let i = 0; i < pts.length; i++){
      const d = Math.hypot(pts[i].x - pos.x, pts[i].y - pos.y);
      if (d < hitDist){ hitDist = d; hit = i; }
    }
    if (hit > 0 && hit < pts.length - 1 && hitDist < 30){
      pts.splice(hit, 1);
      pushHistory();
      drawCurveGraph();
      requestRender();
    }
  });
}

/* ---------------- preview interaction ---------------- */
function initPreviewInteraction(){
  let drag = null;
  previewCanvas.addEventListener('pointerdown', (e) => {
    if (!state.selected) return;
    const r = previewCanvas.getBoundingClientRect();
    const cx = (e.clientX - r.left) / r.width * previewCanvas.width;
    const cy = (e.clientY - r.top) / r.height * previewCanvas.height;
    const W = previewCanvas.width, H = previewCanvas.height;
    const c = selClip();
    if (!c) return;

    if (state.selected.track === 'text'){
      const bb = textBBox(c, W, H);
      if (cx >= bb.x - 12 && cx <= bb.x + bb.w + 12 && cy >= bb.y - 12 && cy <= bb.y + bb.h + 12){
        drag = {
          type:'text', id:c.id,
          startX: e.clientX, startY: e.clientY,
          x0: c.x, y0: c.y
        };
        previewCanvas.setPointerCapture && previewCanvas.setPointerCapture(e.pointerId);
        e.preventDefault();
      }
    } else if (state.selected.track === 'overlay'){
      const m = getMedia(c.mediaId);
      if (!m) return;
      const tr = c.transform;
      const sw = m.width || 1, sh = m.height || 1;
      const base = tr.fit === 'cover' ? Math.max(W/sw, H/sh) : Math.min(W/sw, H/sh);
      const scale = base * (tr.scale || 1) * 0.7;
      const w = sw * scale, h = sh * scale;
      const cxp = W/2 + (tr.x || 0) * W/2, cyp = H/2 + (tr.y || 0) * H/2;
      const rad = -(tr.rotate || 0) * Math.PI / 180;
      const dx = cx - cxp, dy = cy - cyp;
      const lx = dx * Math.cos(rad) - dy * Math.sin(rad);
      const ly = dx * Math.sin(rad) + dy * Math.cos(rad);
      const distHandle = Math.hypot(lx - w/2, ly - h/2);
      if (distHandle < 26){
        drag = {
          type:'resize', id:c.id,
          startX: e.clientX, startY: e.clientY,
          scale0: tr.scale || 1,
          diag0: Math.max(20, Math.hypot(lx, ly))
        };
        previewCanvas.setPointerCapture && previewCanvas.setPointerCapture(e.pointerId);
        e.preventDefault();
      } else if (Math.abs(lx) <= w/2 + 8 && Math.abs(ly) <= h/2 + 8){
        drag = {
          type:'overlay', id:c.id,
          startX: e.clientX, startY: e.clientY,
          x0: tr.x || 0, y0: tr.y || 0
        };
        previewCanvas.setPointerCapture && previewCanvas.setPointerCapture(e.pointerId);
        e.preventDefault();
      }
    }
  });

  previewCanvas.addEventListener('pointermove', (e) => {
    if (!drag) return;
    e.preventDefault();
    const c = selClip();
    if (!c || c.id !== drag.id) return;
    const r = previewCanvas.getBoundingClientRect();
    const dx = (e.clientX - drag.startX) / r.width;
    const dy = (e.clientY - drag.startY) / r.height;
    if (drag.type === 'text'){
      c.x = clamp(drag.x0 + dx, -0.2, 1.2);
      c.y = clamp(drag.y0 + dy, -0.2, 1.2);
      $('#tX').value = Math.round(c.x * 100); $('#tXV').textContent = Math.round(c.x * 100);
      $('#tY').value = Math.round(c.y * 100); $('#tYV').textContent = Math.round(c.y * 100);
    } else if (drag.type === 'overlay'){
      c.transform.x = clamp(drag.x0 + dx * 2, -3, 3);
      c.transform.y = clamp(drag.y0 + dy * 2, -3, 3);
      $('#ePosX').value = Math.round(c.transform.x * 100);
      $('#ePosXV').textContent = Math.round(c.transform.x * 100);
      $('#ePosY').value = Math.round(c.transform.y * 100);
      $('#ePosYV').textContent = Math.round(c.transform.y * 100);
    } else if (drag.type === 'resize'){
      const cur = Math.hypot(e.clientX - drag.startX, e.clientY - drag.startY);
      const k = 1 + cur / Math.max(80, r.width * 0.7);
      c.transform.scale = clamp(drag.scale0 * k, 0.05, 8);
      $('#eScale').value = Math.round(c.transform.scale * 100);
      $('#eScaleV').textContent = Math.round(c.transform.scale * 100);
    }
    requestRender();
  });

  const endDrag = () => {
    if (drag){
      drag = null;
      pushHistory();
      renderTimeline();
    }
  };
  previewCanvas.addEventListener('pointerup', endDrag);
  previewCanvas.addEventListener('pointercancel', endDrag);
}

      version: 2,
      project: state.project,
      media: state.media.map(m => ({
        id: m.id, name: m.name, type: m.type, duration: m.duration,
        width: m.width, height: m.height, offline: true, thumb: null
      }))
    };
    localStorage.setItem('pro_video_editor_project', JSON.stringify(data));
    toast('Project saved. Media files must be re-imported next time.');
  } catch(e){
    toast('Could not save project (storage full?)');
  }
}
function loadProject(){
  let raw = null;
  try { raw = localStorage.getItem('pro_video_editor_project'); } catch(e){}
  if (!raw){ toast('No saved project found'); return; }
  try {
    const data = JSON.parse(raw);
    for (const m of state.media){ if (m.url) URL.revokeObjectURL(m.url); }
    for (const id of Array.from(audioEls.keys())) dropAudioEl(id);
    state.media = (data.media || []).map(m => Object.assign({}, m, { url:null, file:null, offline:true, imgEl:null }));
    state.project = data.project || newProject();
    for (const c of state.project.video){
      const def = defaultEffects();
      c.effects = Object.assign(def, c.effects || {});
    }
    state.selected = null;
    state.playhead = 0;
    state.baseFrame = null;
    state.effectThumbs = null;
    try { mainVideo.removeAttribute('src'); mainVideo.dataset.mediaId = ''; } catch(e){}
    pushHistory();
    refreshAll();
    renderFilterCards();
    buildEffectCards();
    toast('Project loaded. Re-import the original media files to see them.');
  } catch(e){
    toast('Saved project is corrupted');
  }
}

/* ---------------- export ---------------- */
function pickMime(){
  const candidates = [
    'video/webm;codecs=vp9,opus',
    'video/webm;codecs=vp8,opus',
    'video/webm;codecs=vp9',
    'video/webm;codecs=vp8',
    'video/webm',
    'video/mp4;codecs=avc1.42E01E,mp4a.40.2',
    'video/mp4'
  ];
  for (const c of candidates){
    try { if (window.MediaRecorder && MediaRecorder.isTypeSupported(c)) return c; } catch(e){}
  }
  return '';
}
function checkCompat(){
  const notes = [];
  if (!window.MediaRecorder) notes.push('MediaRecorder is unavailable — video export is disabled in this browser/WebView.');
  if (!HTMLCanvasElement.prototype.captureStream) notes.push('canvas.captureStream() is unavailable — video export is disabled.');
  const el = $('#compatNote');
  el.innerHTML = notes.length ? '<div class="warn">' + notes.map(escapeHtml).join('<br>') + '</div>' : '';
  const ex = $('#expCompat');
  const expNotes = [];
  if (!window.MediaRecorder) expNotes.push('MediaRecorder not available.');
  if (!HTMLCanvasElement.prototype.captureStream) expNotes.push('canvas.captureStream() not available.');
  if (!pickMime()) expNotes.push('No supported recording codec found.');
  ex.innerHTML = expNotes.length ? '<div class="err">' + expNotes.map(escapeHtml).join('<br>') + '</div>' : '';
  $('#expStart').disabled = expNotes.length > 0;
}
function openExport(){
  checkCompat();
  const dur = projectDuration();
  $('#expInfo').textContent = dur > 0
    ? 'Duration ' + fmtTime(dur) + ' — filters, effects, color, text and image overlays will be rendered into the exported video.'
    : 'Timeline is empty. Add media first.';
  $('#expBar').style.width = '0%';
  $('#expStatus').textContent = 'Ready.';
  $('#expResult').innerHTML = '';
  $('#exportModal').classList.add('open');
}
function closeExport(){
  if (state.exporting) return;
  $('#exportModal').classList.remove('open');
}

async function startExport(){
  if (state.exporting) return;
  const dur = projectDuration();
  if (dur <= 0){ toast('Timeline is empty'); return; }
  const mime = pickMime();
  if (!window.MediaRecorder || !HTMLCanvasElement.prototype.captureStream || !mime){
    $('#expStatus').textContent = 'Export not supported in this browser.';
    return;
  }

  const res = parseInt($('#expRes').value, 10);
  const fps = parseInt($('#expFps').value, 10);
  const bitrate = parseInt($('#expQuality').value, 10);
  const aspect = aspectValue(state.project.aspect);
  let outH = res;
  let outW = Math.round(outH * aspect / 2) * 2;
  if (outW > 3840){ outW = 3840; outH = Math.round(outW / aspect / 2) * 2; }

  const canvas = document.createElement('canvas');
  canvas.width = outW;
  canvas.height = outH;
  const ectx = canvas.getContext('2d', { alpha:false });

  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    if (audioCtx.state === 'suspended') await audioCtx.resume();
  } catch(e){
    $('#expStatus').textContent = 'AudioContext unavailable — exporting without audio.';
  }

  let dest = null;
  if (audioCtx){
    try {
      dest = audioCtx.createMediaStreamDestination();
      if (!videoSrcNode){
        videoSrcNode = audioCtx.createMediaElementSource(mainVideo);
        videoSrcNode.connect(audioCtx.destination);
      }
      videoSrcNode.connect(dest);
      for (const ac of state.project.audio){
        const m = getMedia(ac.mediaId);
        if (!m || !m.url) continue;
        const el = getAudioEl(ac);
        let node = audioSrcNodes.get(ac.id);
        if (!node){
          node = audioCtx.createMediaElementSource(el);
          node.connect(audioCtx.destination);
          audioSrcNodes.set(ac.id, node);
        }
        node.connect(dest);
      }
    } catch(e){
      dest = null;
    }
  }

  const stream = canvas.captureStream(fps);
  if (dest){
    dest.stream.getAudioTracks().forEach(tr => stream.addTrack(tr));
  }

  let rec;
  try {
    rec = new MediaRecorder(stream, {
      mimeType: mime,
      videoBitsPerSecond: bitrate,
      audioBitsPerSecond: 128000
    });
  } catch(e){
    $('#expStatus').textContent = 'MediaRecorder failed to start: ' + e.message;
    return;
  }

  const chunks = [];
  rec.ondataavailable = (e) => { if (e.data && e.data.size) chunks.push(e.data); };

  state.exporting = true;
  state.exportCancelled = false;
  $('#expStart').disabled = true;
  $('#expClose').disabled = true;
  $('#expStatus').textContent = 'Recording…';

  const startWall = performance.now();
  let exportRaf = 0;
  let lastExportTs = 0;

  const finishExport = (cancelled) => {
    state.exporting = false;
    $('#expStart').disabled = false;
    $('#expClose').disabled = false;
    try { mainVideo.pause(); } catch(e){}
    for (const el of audioEls.values()){ try { el.pause(); } catch(e){} }
    if (cancelled){
      $('#expStatus').textContent = 'Export cancelled.';
      $('#expBar').style.width = '0%';
      return;
    }
  };

  rec.onstop = () => {
    const elapsed = ((performance.now() - startWall) / 1000).toFixed(1);
    if (state.exportCancelled){
      finishExport(true);
      return;
    }
    const blob = new Blob(chunks, { type: mime.split(';')[0] });
    const url = URL.createObjectURL(blob);
    const ext = mime.indexOf('mp4') >= 0 ? 'mp4' : 'webm';
    $('#expBar').style.width = '100%';
    $('#expStatus').textContent = 'Export finished in ' + elapsed + 's · ' + (blob.size/1048576).toFixed(2) + ' MB';
    $('#expResult').innerHTML =
      '<div class="group" style="margin-top:10px">' +
      '<a class="btn primary" style="width:100%;text-decoration:none" download="export.' + ext + '" href="' + url + '">⬇ Download Video</a>' +
      '<video src="' + url + '" controls style="width:100%;margin-top:8px;border-radius:8px;background:#000"></video>' +
      '</div>';
    finishExport(false);
  };

  pause();
  state.playhead = 0;
  state.activeClipId = null;
  updatePlayheadUI();

  ensureVideoFrame(0);
  await new Promise(r => setTimeout(r, 120));
  renderFrame(ectx, outW, outH, 0, false);

  state.playing = true;
  state.lastTs = 0;

  try { rec.start(250); } catch(e){
    $('#expStatus').textContent = 'Could not start recording: ' + e.message;
    finishExport(true);
    return;
  }

  const exportLoop = (ts) => {
    if (!state.exporting) return;
    if (!lastExportTs) lastExportTs = ts;
    const dt = Math.min(0.25, (ts - lastExportTs) / 1000);
    lastExportTs = ts;

    state.playhead += dt;
    if (state.playhead >= dur){
      state.playhead = dur;
      renderFrame(ectx, outW, outH, Math.max(0, dur - 0.001), false);
      blitToPreview(canvas);
      updatePlayheadUI();
      state.exporting = false;
      try { rec.stop(); } catch(e){ finishExport(false); }
      return;
    }

    const clip = clipAt(state.playhead);
    if (!clip){
      if (state.activeClipId){ state.activeClipId = null; try { mainVideo.pause(); } catch(e){} }
    } else {
      if (state.activeClipId !== clip.id){
        state.activeClipId = clip.id;
        loadClipForPlayback(clip, state.playhead);
      } else {
        const target = clip.inPoint + (state.playhead - clip.start) * (clip.speed || 1);
        if (!mainVideo.seeking && Math.abs(mainVideo.currentTime - target) > 0.32){
          try { mainVideo.currentTime = target; } catch(e){}
        }
        if (mainVideo.paused) mainVideo.play().catch(()=>{});
      }
    }

    for (const ac of state.project.audio){
      const m = getMedia(ac.mediaId);
      if (!m || !m.url) continue;
      const el = getAudioEl(ac);
      const t0 = ac.start, t1 = ac.start + clipDur(ac);
      if (state.playhead >= t0 && state.playhead < t1){
        const local = state.playhead - t0;
        const target = clamp(ac.inPoint + local * (ac.speed || 1), 0, Math.max(0, (m.duration||0) - 0.03));
        if (el.paused){ try { el.currentTime = target; } catch(e){} el.play().catch(()=>{}); }
        el.volume = clamp(audioVolumeAt(ac, local), 0, 1);
        el.muted = !!ac.muted;
      } else if (!el.paused){ try { el.pause(); } catch(e){} }
    }
    if (clip){
      mainVideo.volume = clamp((clip.muted ? 0 : clip.volume) * state.project.originalVolume, 0, 1);
      mainVideo.muted = state.project.originalMuted || clip.muted;
    }

    renderFrame(ectx, outW, outH, state.playhead, false);
    blitToPreview(canvas);
    updatePlayheadUI();

    const pct = (state.playhead / dur) * 100;
    $('#expBar').style.width = pct.toFixed(1) + '%';
    $('#expStatus').textContent = 'Recording… ' + pct.toFixed(0) + '% · ' +
      ((performance.now() - startWall)/1000).toFixed(1) + 's elapsed';

    exportRaf = requestAnimationFrame(exportLoop);
  };
  exportRaf = requestAnimationFrame(exportLoop);

  window.__exportCancel = () => {
    state.exportCancelled = true;
    state.exporting = false;
    cancelAnimationFrame(exportRaf);
    try { rec.stop(); } catch(e){}
    finishExport(true);
  };
}

function blitToPreview(srcCanvas){
  const ctx = previewCanvas.getContext('2d');
  const W = previewCanvas.width, H = previewCanvas.height;
  ctx.setTransform(1,0,0,1,0,0);
  ctx.filter = 'none';
  ctx.globalAlpha = 1;
  ctx.globalCompositeOperation = 'source-over';
  ctx.fillStyle = '#000';
  ctx.fillRect(0, 0, W, H);
  const ar = srcCanvas.width / srcCanvas.height;
  let w = W, h = W / ar;
  if (h > H){ h = H; w = H * ar; }
  ctx.drawImage(srcCanvas, (W - w)/2, (H - h)/2, w, h);
}

    if (!c || !c.effects) return;
    const key = inp.dataset.param;
    const v = parseFloat(inp.value);
    c.effects[key] = v;
    const valEl = inp.parentElement.querySelector('[data-paramval]');
    if (valEl) valEl.textContent = Math.round(v);
    const card = $('#effectCards .effect-card[data-effect="' + key + '"]');
    if (card) card.classList.toggle('active', v > 0);
    requestRender();
  });
  $('#effectsPanel').addEventListener('change', (e) => {
    const inp = e.target;
    if (inp.matches('input[type=range][data-param]')){
      pushHistory();
      const c = selClip();
      if (c && c.effects && (c.effects[inp.dataset.param] || 0) <= 0){
        buildEffectsSliders();
      }
    }
  });

  /* ---- curve ---- */
  $('#curveTabs').addEventListener('click', (e) => {
    const b = e.target.closest('button');
    if (!b) return;
    state.activeCurve = b.dataset.ch;
    $$('#curveTabs button').forEach(x => x.classList.toggle('active', x === b));
    drawCurveGraph();
  });
  $('#cvReset').addEventListener('click', () => {
    const c = currentCurvesClip();
    if (!c) return toast('Select a video clip');
    c.curves[state.activeCurve] = [{x:0,y:0},{x:255,y:255}];
    pushHistory(); drawCurveGraph(); requestRender();
  });
  $('#cvResetAll').addEventListener('click', () => {
    const c = currentCurvesClip();
    if (!c) return toast('Select a video clip');
    c.curves = defaultCurves();
    pushHistory(); drawCurveGraph(); requestRender();
  });

  /* ---- crop ---- */
  const cropBind = (id, key) => {
    const el = $('#' + id);
    el.addEventListener('input', () => {
      const c = selClip();
      if (!c || !c.crop) return;
      const v = parseFloat(el.value);
      const other = key === 'top' ? 'bottom' : key === 'bottom' ? 'top' : key === 'left' ? 'right' : 'left';
      if (c.crop[key] + c.crop[other] > 90) c.crop[key] = 90 - c.crop[other];
      else c.crop[key] = v;
      el.value = c.crop[key];
      $('#' + id + 'V').textContent = Math.round(c.crop[key]);
      requestRender();
    });
    el.addEventListener('change', () => pushHistory());
  };
  cropBind('crTop','top'); cropBind('crBottom','bottom');
  cropBind('crLeft','left'); cropBind('crRight','right');
  $('#crReset').addEventListener('click', () => {
    const c = selClip();
    if (!c || !c.crop) return toast('Select a video or image clip');
    c.crop = defaultCrop();
    pushHistory(); syncControls(); requestRender();
  });
  const cropAspect = (ar) => {
    const c = selClip();
    if (!c || !c.crop) return toast('Select a video or image clip');
    const m = getMedia(c.mediaId);
    if (!m || !m.width || !m.height) return toast('Media dimensions unknown');
    const srcAR = m.width / m.height;
    let cropL = 0, cropT = 0;
    if (srcAR > ar){
      const keep = ar / srcAR;
      cropL = (1 - keep) / 2 * 100;
    } else {
      const keep = srcAR / ar;
      cropT = (1 - keep) / 2 * 100;
    }
    c.crop = { top: cropT, bottom: cropT, left: cropL, right: cropL };
    pushHistory(); syncControls(); requestRender();
  };
  $('#crFit16').addEventListener('click', () => cropAspect(16/9));
  $('#crFit9').addEventListener('click', () => cropAspect(9/16));
  $('#crFit1').addEventListener('click', () => cropAspect(1));

  /* ---- speed ---- */
  $('#speedGrid').addEventListener('click', (e) => {
    const b = e.target.closest('button');
    if (!b) return;
    const c = selClip();
    if (!c || (state.selected.track !== 'video' && state.selected.track !== 'audio')){
      toast('Select a video or audio clip'); return;
    }
    c.speed = parseFloat(b.dataset.speed);
    pushHistory(); refreshAll();
  });
  bindRange('spCustom', v => {
    const c = selClip();
    if (!c) return;
    c.speed = v / 100;
    $('#spCustomV').textContent = (v / 100).toFixed(2);
    $$('#speedGrid .btn').forEach(b => b.classList.toggle('active', Math.abs(parseFloat(b.dataset.speed) - c.speed) < 0.001));
    renderTimeline();
  });

  /* ---- export modal ---- */
  $('#expClose').addEventListener('click', closeExport);
  $('#expStart').addEventListener('click', startExport);
  $('#exportModal').addEventListener('click', (e) => {
    if (e.target === $('#exportModal')) closeExport();
  });

  /* ---- video element events ---- */
  mainVideo.addEventListener('seeked', () => { if (!state.playing && !state.exporting) requestRender(); });
  mainVideo.addEventListener('loadeddata', () => { if (!state.playing && !state.exporting) requestRender(); });
  mainVideo.addEventListener('error', () => {
    if (mainVideo.src) toast('This media could not be played by your browser.');
  });

  /* ---- misc ---- */
  window.addEventListener('resize', () => { resizePreview(); });
  window.addEventListener('orientationchange', () => setTimeout(resizePreview, 250));
  document.addEventListener('visibilitychange', () => {
    if (document.hidden && state.playing && !state.exporting) pause();
  });
  window.addEventListener('beforeunload', () => {
    for (const m of state.media){ if (m.url) URL.revokeObjectURL(m.url); }
  });
  window.addEventListener('keydown', (e) => {
    if (e.target.matches('input,textarea,select')) return;
    if (e.code === 'Space'){ e.preventDefault(); state.playing ? pause() : play(); }
    else if (e.code === 'ArrowRight'){ state.playhead += e.shiftKey ? 1 : 1/30; updatePlayheadUI(); if(!state.playing) requestRender(); }
    else if (e.code === 'ArrowLeft'){ state.playhead = Math.max(0, state.playhead - (e.shiftKey ? 1 : 1/30)); updatePlayheadUI(); if(!state.playing) requestRender(); }
    else if (e.key === 's' && (e.ctrlKey || e.metaKey)){ e.preventDefault(); saveProject(); }
  });

  updateHistoryButtons();
  $('#previewEmpty').style.display = 'block';
}

/* ---------------- clip operations ---------------- */
function splitAtPlayhead(){
  const c = selClip();
  if (!c){ toast('Select a clip to split'); return; }
  const track = state.selected.track;
  const t = state.playhead;
  const end = clipEndOf(track, c);
  if (t <= c.start + 0.05 || t >= end - 0.05){ toast('Move the playhead inside the clip'); return; }

  if (track === 'video' || track === 'audio'){
    const sp = c.speed || 1;
    const srcSplit = c.inPoint + (t - c.start) * sp;
    const right = JSON.parse(JSON.stringify(c));
    right.id = uid();
    right.start = t;
    right.inPoint = srcSplit;
    c.outPoint = srcSplit;
    const arr = state.project[track];
    const i = arr.indexOf(c);
    arr.splice(i + 1, 0, right);
    state.selected = { track, id: right.id };
  } else {
    const dt = t - c.start;
    const right = JSON.parse(JSON.stringify(c));
    right.id = uid();
    right.start = t;
    right.duration = c.duration - dt;
    c.duration = dt;
    const arr = state.project[track];
    const i = arr.indexOf(c);
    arr.splice(i + 1, 0, right);
    state.selected = { track, id: right.id };
  }
  pushHistory();
  refreshAll();
  toast('Clip split');
}
function duplicateSelected(){
  const c = selClip();
  if (!c){ toast('Select a clip'); return; }
  const track = state.selected.track;
  const cp = JSON.parse(JSON.stringify(c));
  cp.id = uid();
  cp.start = clipEndOf(track, c);
  state.project[track].push(cp);
  state.selected = { track, id: cp.id };
  pushHistory();
  refreshAll();
  toast('Clip duplicated');
}
function deleteSelected(){
  const c = selClip();
  if (!c){ toast('Select a clip'); return; }
  const track = state.selected.track;
  state.project[track] = state.project[track].filter(x => x.id !== c.id);
  if (track === 'audio') dropAudioEl(c.id);
  state.selected = null;
  pushHistory();
  refreshAll();
  toast('Clip deleted');
}
function setTrimStart(){
  const c = selClip();
  if (!c){ toast('Select a clip'); return; }
  const track = state.selected.track;
  if (track === 'text'){ toast('Trim is not available for text'); return; }
  const t = clamp(state.playhead, c.start, clipEndOf(track, c) - 0.05);
  if (track === 'overlay'){
    const cut = t - c.start;
    c.duration -= cut;
    c.start = t;
  } else {
    const sp = c.speed || 1;
    const src = c.inPoint + (t - c.start) * sp;
    if (src >= c.outPoint - 0.05){ toast('Too close to the end'); return; }
    c.inPoint = src;
    c.start = t;
  }
  pushHistory();
  refreshAll();
}
function setTrimEnd(){
  const c = selClip();
  if (!c){ toast('Select a clip'); return; }
  const track = state.selected.track;
  if (track === 'text'){ toast('Trim is not available for text'); return; }
  const t = clamp(state.playhead, c.start + 0.05, Math.max(projectDuration(), clipEndOf(track, c)));
  if (track === 'overlay'){
    c.duration = Math.max(0.2, t - c.start);
  } else {
    const sp = c.speed || 1;
    const src = c.inPoint + (t - c.start) * sp;
    if (src <= c.inPoint + 0.05){ toast('Too close to the start'); return; }
    const m = getMedia(c.mediaId);
    c.outPoint = clamp(src, c.inPoint + 0.05, m && m.duration ? m.duration : src);
  }
  pushHistory();
  refreshAll();
}
